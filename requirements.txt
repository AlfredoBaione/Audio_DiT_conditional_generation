# =============================================================================
# Conditioned Audio DiT — Python dependencies
#
# Versions are intentionally left unpinned: PyTorch must match the CUDA
# toolkit available on each machine (IRCAM servers differ), and the rest floats
# with it. After you have a working environment, run `pip freeze > requirements.lock.txt`
# to pin an exact, reproducible set for that specific server.
#
# Install order:
#   1) PyTorch stack matching your CUDA (see README — do NOT rely on the plain
#      `pip install torch` line below on GPU servers).
#   2) pip install -r requirements.txt
#
# This installs the FULL conditioning stack (every extractor), not just the
# conditions currently enabled in the config: the extractors are imported
# lazily, so having them installed costs nothing at runtime and means switching
# `conditioning.enabled_frame` never requires touching the environment.
# =============================================================================

# ---- core numerics / audio I/O ----
numpy
scipy
soundfile
librosa
tqdm
matplotlib
omegaconf
tensorboard         # torch.utils.tensorboard SummaryWriter (training/test logging)

# ---- PyTorch stack ----
# On a GPU server install these matching your CUDA FIRST, e.g.:
#   pip install torch torchaudio torchvision --index-url https://download.pytorch.org/whl/cu121
# The bare names below are a CPU/fallback default only.
torch
torchaudio
torchvision

# ---- DAC codec (imported in code as `dac`) ----
descript-audio-codec

# ---- conditioning extractors (frame-level) ----
torchcrepe           # f0     — monophonic pitch contour (CREPE)
basic-pitch          # melody — 88-bin note posteriorgram
                     #   NB: pulls TensorFlow. On a conda base that already ships
                     #   TF (e.g. IRCAM tf2.18) pip may downgrade protobuf and warn
                     #   about a TF conflict. Harmless for THIS pipeline (the code
                     #   sets USE_TF=0 and never imports TF), but if TensorFlow
                     #   itself matters to you on that machine, install this in a
                     #   separate env. Verify after installing:
                     #     python -c "from torch.utils.tensorboard import SummaryWriter; print('tb ok')"
                     #     python -c "import dac; print('dac ok')"
beat_this @ git+https://github.com/CPJKU/beat_this.git   # rhythm — beat/downbeat (not on PyPI)

# ---- conditioning encoders (global) ----
transformers         # CLAP (text) + CLIP (image) global conditions
Pillow               # image loading for the CLIP image condition

# ---- evaluation metrics ----
mir_eval             # melody RPA/RCA fidelity (condition_metrics.py)
encodec              # Encodec embeddings (audio-FAD path)
# VGGish is loaded via torch.hub (harritaylor/torchvggish) — no pip package.
